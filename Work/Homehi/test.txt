  <script>
    // ========== 存储层 ==========
    const STORAGE_KEYS = {
      PROFILES: 'lc_profiles',
      ACTIVE_PROFILE: 'lc_active_profile',
      CONVERSATIONS: 'lc_conversations',
      ACTIVE_CONV: 'lc_active_conv',
      TODO_GROUPS: 'lc_todo_groups',
      TODO_ITEMS: 'lc_todo_items',
      ACTIVE_TODO_GROUP: 'lc_active_todo_group',
      SETTINGS: 'lc_settings'
    };

    const PRESETS = {
      openai: { 
        name: 'OpenAI 默认', 
        baseUrl: 'https://api.openai.com/v1', 
        model: 'gpt-4o-mini',
        systemPrompt: '你是一个有帮助的 AI 助手，请用简洁清晰的语言回答用户的问题。'
      },
      deepseek: { 
        name: 'DeepSeek', 
        baseUrl: 'https://api.deepseek.com/v1', 
        model: 'deepseek-chat',
        systemPrompt: '你是一个专业的 AI 助手，请准确回答用户的问题。'
      },
      zhipu: { 
        name: '智谱 AI', 
        baseUrl: 'https://open.bigmodel.cn/api/paas/v4', 
        model: 'glm-4-flash',
        systemPrompt: '你是一个智能助手，请帮助用户解决问题。'
      },
      translator: {
        name: '📚 翻译助手',
        baseUrl: 'https://api.openai.com/v1',
        model: 'gpt-4o-mini',
        systemPrompt: '你是一位专业的翻译专家。请将用户输入的所有内容准确翻译成中文（如果输入是中文则翻译成英文）。要求：\n1. 保持原文语气和风格\n2. 专业术语准确\n3. 译文自然流畅，符合目标语言表达习惯\n4. 只输出翻译结果，不要添加额外解释'
      },
      local: { 
        name: '本地代理', 
        baseUrl: 'http://localhost:8080/v1', 
        model: 'gpt-4o',
        systemPrompt: '你是一个有帮助的 AI 助手。'
      },
      custom: { name: '自定义', baseUrl: '', model: '', systemPrompt: '' }
    };

    // ========== 全局状态 ==========
    let profiles = [];
    let activeProfileId = null;
    let conversations = [];
    let activeConvId = null;
    let todoGroups = [];
    let todoItems = [];
    let activeTodoGroupId = null;
    let isStreaming = false;
    let currentTab = 'chat';

    // ========== 初始化 ==========
    function init() {
      lucide.createIcons();
      
      // 加载所有数据
      profiles = storeGet(STORAGE_KEYS.PROFILES, []);
      activeProfileId = storeGet(STORAGE_KEYS.ACTIVE_PROFILE, null);
      conversations = storeGet(STORAGE_KEYS.CONVERSATIONS, []);
      activeConvId = storeGet(STORAGE_KEYS.ACTIVE_CONV, null);
      todoGroups = storeGet(STORAGE_KEYS.TODO_GROUPS, []);
      todoItems = storeGet(STORAGE_KEYS.TODO_ITEMS, []);
      activeTodoGroupId = storeGet(STORAGE_KEYS.ACTIVE_TODO_GROUP, null);

      // 初始化默认待办分组
      if (todoGroups.length === 0) {
        const defaultGroup = { id: genId(), name: '默认分组', collapsed: false, createdAt: Date.now() };
        todoGroups.push(defaultGroup);
        activeTodoGroupId = defaultGroup.id;
        storeSet(STORAGE_KEYS.TODO_GROUPS, todoGroups);
        storeSet(STORAGE_KEYS.ACTIVE_TODO_GROUP, activeTodoGroupId);
      }

      if (profiles.length === 0) {
        document.getElementById('welcomeModal').classList.remove('hidden');
      } else {
        renderProfileSelector();
        renderConversationList();
        renderTodoGroups();
        renderTodoItems();
        if (activeConvId) loadConversation(activeConvId);
      }
    }

    function storeGet(key, def) {
      try {
        const v = localStorage.getItem(key);
        return v ? JSON.parse(v) : def;
      } catch { return def; }
    }

    function storeSet(key, val) {
      localStorage.setItem(key, JSON.stringify(val));
    }

    // ========== Tab 切换 ==========
    function switchTab(tab) {
      currentTab = tab;
      document.getElementById('tabChat').className = tab === 'chat' 
        ? 'flex-1 py-3 text-sm font-medium tab-active' 
        : 'flex-1 py-3 text-sm font-medium text-gray-400 hover:text-white';
      document.getElementById('tabTodo').className = tab === 'todo' 
        ? 'flex-1 py-3 text-sm font-medium tab-active' 
        : 'flex-1 py-3 text-sm font-medium text-gray-400 hover:text-white';
      
      document.getElementById('chatPanel').classList.toggle('hidden', tab !== 'chat');
      document.getElementById('todoPanel').classList.toggle('hidden', tab !== 'todo');
      document.getElementById('chatMain').classList.toggle('hidden', tab !== 'chat');
      document.getElementById('todoMain').classList.toggle('hidden', tab !== 'todo');
      
      lucide.createIcons();
    }

    // ========== 首次引导 ==========
    function updateWelcomeFields() {
      const p = document.getElementById('welcomeProvider').value;
      const preset = PRESETS[p];
      if (preset) {
        document.getElementById('welcomeBaseUrl').value = preset.baseUrl;
        document.getElementById('welcomePrompt').value = preset.systemPrompt || '';
      }
    }

    function completeOnboarding() {
      const token = document.getElementById('welcomeToken').value.trim();
      const baseUrl = document.getElementById('welcomeBaseUrl').value.trim();
      const prompt = document.getElementById('welcomePrompt').value.trim();
      const provider = document.getElementById('welcomeProvider').value;

      if (!token) { alert('请输入 API Token'); return; }
      if (!baseUrl) { alert('请输入接口地址'); return; }

      const profile = {
        id: genId(),
        name: PRESETS[provider]?.name || '默认配置',
        provider: provider,
        apiKey: token,
        baseUrl: baseUrl,
        model: PRESETS[provider]?.model || 'gpt-4o-mini',
        systemPrompt: prompt || '你是一个有帮助的 AI 助手。',
        temperature: 0.7
      };

      profiles = [profile];
      activeProfileId = profile.id;
      storeSet(STORAGE_KEYS.PROFILES, profiles);
      storeSet(STORAGE_KEYS.ACTIVE_PROFILE, activeProfileId);

      document.getElementById('welcomeModal').classList.add('hidden');
      renderProfileSelector();
      newConversation();
    }

    // ========== 配置档案管理 (类 cc-switch) ==========

    function renderMessages() {
      const container = document.getElementById('messagesContainer');
      const conv = getActiveConv();

      if (!conv || conv.messages.length === 0) {
        container.innerHTML = `
          <div class="text-center text-gray-500 mt-20">

    function createMessageEl(msg) {
      const isUser = msg.role === 'user';
      const wrap = document.createElement('div');
      wrap.className = `flex ${isUser ? 'justify-end' : 'justify-start'}`;

      const bubble = document.createElement('div');
      bubble.className = `max-w-3xl rounded-2xl px-4 py-3 ${
        isUser ? 'bg-primary text-white rounded-br-md' : 'bg-gray-700 text-gray-100 rounded-bl-md'
      }`;

      if (isUser) {
        bubble.textContent = msg.content;
      } else {
        bubble.className += ' message-content';
        bubble.innerHTML = marked.parse(msg.content);
        bubble.querySelectorAll('pre code').forEach(block => {
          hljs.highlightElement(block);
        });
      }

      wrap.appendChild(bubble);
      return wrap;
    }

    async function sendMessage() {
      if (isStreaming) return;
      const input = document.getElementById('messageInput');
      const content = input.value.trim();
      if (!content) return;

      const profile = getActiveProfile();
      if (!profile) { alert('请先配置 API'); return; }

      let conv = getActiveConv();
      if (!conv) { newConversation(); conv = getActiveConv(); }

      // 添加用户消息
      conv.messages.push({ role: 'user', content });
      if (conv.messages.length === 1) {
        conv.title = content.slice(0, 20) + (content.length > 20 ? '...' : '');
      }
      conv.updatedAt = Date.now();
      saveConversations();
      renderConversationList();
      renderMessages();
      input.value = '';
      autoResize(input);

      // 添加空助手消息
      const assistantMsg = { role: 'assistant', content: '' };
      conv.messages.push(assistantMsg);

      const container = document.getElementById('messagesContainer');
      const msgEl = createMessageEl({ role: 'assistant', content: '' });
      const contentEl = msgEl.querySelector('.message-content');
      contentEl.classList.add('typing-cursor');
      container.appendChild(msgEl);
      container.scrollTop = container.scrollHeight;

      isStreaming = true;
      document.getElementById('sendBtn').disabled = true;

      try {
        // 构建消息：优先使用对话独立的 System Prompt，确保在最前面
        const messages = [];
        const sysPrompt = conv.systemPrompt || profile.systemPrompt;
        if (sysPrompt) {
          messages.push({ role: 'system', content: sysPrompt });
        }
        
        // 添加历史消息（排除 system，避免重复）
        conv.messages.slice(0, -1).forEach(m => {
          if (m.role !== 'system') messages.push(m);
        });

        const response = await fetch(`${profile.baseUrl}/chat/completions`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${profile.apiKey}`
          },
          body: JSON.stringify({
            model: profile.model,
            messages: messages,
            stream: true,
            temperature: profile.temperature || 0.7
          })
        });

        if (!response.ok) {
          const errText = await response.text();
          throw new Error(`请求失败 (${response.status}): ${errText}`);
        }

        const reader = response.body.getReader();
        const decoder = new TextDecoder();
        let buffer = '';

        while (true) {
          const { done, value } = await reader.read();
          if (done) break;

          buffer += decoder.decode(value, { stream: true });
          const lines = buffer.split('\n');
          buffer = lines.pop();

          for (const line of lines) {
            const trimmed = line.trim();
            if (!trimmed || !trimmed.startsWith('data: ')) continue;
            const data = trimmed.slice(6);
            if (data === '[DONE]') continue;

            try {
              const json = JSON.parse(data);
              const delta = json.choices?.[0]?.delta?.content;
              if (delta) {
                assistantMsg.content += delta;
                contentEl.innerHTML = marked.parse(assistantMsg.content);
                contentEl.querySelectorAll('pre code').forEach(block => hljs.highlightElement(block));
                container.scrollTop = container.scrollHeight;
              }
            } catch (e) { /* 忽略解析错误 */ }
          }
        }

        contentEl.classList.remove('typing-cursor');
        saveConversations();

      } catch (err) {
        contentEl.classList.remove('typing-cursor');
        contentEl.innerHTML = `<span class="text-red-400">错误: ${err.message}</span>`;
        conv.messages.pop();
        saveConversations();
      }

      isStreaming = false;
      document.getElementById('sendBtn').disabled = false;
    }

    function renderTodoGroups() {
      const list = document.getElementById('todoGroupList');
      list.innerHTML = '';
      
      todoGroups.forEach(group => {
        const itemCount = todoItems.filter(i => i.groupId === group.id).length;
        const div = document.createElement('div');
        div.className = 'bg-gray-700/50 rounded-lg overflow-hidden';
        
        const header = document.createElement('div');
        header.className = 'flex items-center justify-between px-3 py-2 cursor-pointer hover:bg-gray-700';
        header.innerHTML = `
          <div class="flex items-center gap-2 flex-1 min-w-0">
            <i data-lucide="${group.collapsed ? 'chevron-right' : 'chevron-down'}" class="w-4 h-4 text-gray-400 shrink-0"></i>
            <i data-lucide="folder" class="w-4 h-4 text-primary shrink-0"></i>
            <span class="text-sm truncate">${group.name}</span>
            <span class="text-xs text-gray-500 shrink-0">(${itemCount})</span>
          </div>
        `;
        header.onclick = () => toggleGroupCollapse(group.id);
        
        const content = document.createElement('div');
        content.className = group.collapsed ? 'hidden' : 'pb-2 px-2 space-y-1';
        
        // 分组内任务预览
        todoItems.filter(i => i.groupId === group.id).slice(0, 3).forEach(item => {
          const itemDiv = document.createElement('div');
          itemDiv.className = `text-xs px-2 py-1 rounded truncate cursor-pointer hover:bg-gray-600 ${
            item.completed ? 'text-gray-500 line-through' : 'text-gray-300'
          }`;
          itemDiv.textContent = item.content;
          itemDiv.onclick = (e) => {
            e.stopPropagation();
            selectTodoGroup(group.id);
          };
          content.appendChild(itemDiv);
        });

        if (itemCount > 3) {
          const more = document.createElement('div');
          more.className = 'text-xs text-gray-500 px-2 py-1';
          more.textContent = `还有 ${itemCount - 3} 项...`;
          content.appendChild(more);
        }

        div.appendChild(header);
        div.appendChild(content);
        list.appendChild(div);
      });
      
      lucide.createIcons();
    }

    function toggleGroupCollapse(groupId) {
      const group = todoGroups.find(g => g.id === groupId);
      if (group) {
        group.collapsed = !group.collapsed;
        storeSet(STORAGE_KEYS.TODO_GROUPS, todoGroups);
        renderTodoGroups();
      }
    }

    function selectTodoGroup(groupId) {
      activeTodoGroupId = groupId;
      storeSet(STORAGE_KEYS.ACTIVE_TODO_GROUP, groupId);
      renderTodoGroups();
      renderTodoItems();
    }

    function addTodoGroup() {
      const name = prompt('请输入分组名称：');
      if (!name?.trim()) return;
      
      const group = {
        id: genId(),
        name: name.trim(),
        collapsed: false,
        createdAt: Date.now()
      };
      todoGroups.push(group);
      activeTodoGroupId = group.id;
      storeSet(STORAGE_KEYS.TODO_GROUPS, todoGroups);
      storeSet(STORAGE_KEYS.ACTIVE_TODO_GROUP, activeTodoGroupId);
      renderTodoGroups();
      renderTodoItems();
    }

    function renderTodoItems() {
      const list = document.getElementById('todoItemsList');
      const group = todoGroups.find(g => g.id === activeTodoGroupId);
      
      document.getElementById('currentGroupName').textContent = group ? group.name : '选择分组查看待办';
      
      if (!group) {
        list.innerHTML = '<div class="text-center text-gray-500 mt-20">请先选择一个分组</div>';
        return;
      }

      const items = todoItems.filter(i => i.groupId === group.id).sort((a, b) => {
        if (a.completed !== b.completed) return a.completed ? 1 : -1;
        return b.createdAt - a.createdAt;
      });

      if (items.length === 0) {
        list.innerHTML = '<div class="text-center text-gray-500 mt-20">暂无任务，点击右上角添加</div>';
        return;
      }

      list.innerHTML = '';
      items.forEach(item => {
        const div = document.createElement('div');
        div.className = `todo-item flex items-start gap-3 p-3 bg-gray-700/50 rounded-lg group ${
          item.completed ? 'opacity-60' : ''
        }`;
        
        div.innerHTML = `
          <input type="checkbox" ${item.completed ? 'checked' : ''} 
            class="mt-1 w-4 h-4 accent-primary cursor-pointer"
            onchange="toggleTodoComplete('${item.id}')">
          <div class="flex-1 min-w-0">
            <div class="${item.completed ? 'line-through text-gray-500' : 'text-gray-200'} break-words">
              ${escapeHtml(item.content)}
            </div>
          </div>
          <div class="todo-actions flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
            <button onclick="copyTodoContent('${item.id}')" class="p-1 text-gray-400 hover:text-white" title="复制内容">
              <i data-lucide="copy" class="w-4 h-4"></i>
            </button>
            <button onclick="editTodoItem('${item.id}')" class="p-1 text-gray-400 hover:text-white" title="编辑">
              <i data-lucide="edit" class="w-4 h-4"></i>
            </button>
            <button onclick="deleteTodoItem('${item.id}')" class="p-1 text-gray-400 hover:text-red-400" title="删除">
              <i data-lucide="trash-2" class="w-4 h-4"></i>
            </button>
          </div>
        `;
        list.appendChild(div);
      });
      
      lucide.createIcons();
    }

    function addTodoItem() {
      if (!activeTodoGroupId) {
        alert('请先选择一个分组');
        return;
      }
      const content = prompt('请输入待办内容：');
      if (!content?.trim()) return;
      
      todoItems.push({
        id: genId(),
        groupId: activeTodoGroupId,
        content: content.trim(),
        completed: false,
        createdAt: Date.now()
      });
      
      storeSet(STORAGE_KEYS.TODO_ITEMS, todoItems);
      renderTodoGroups();
      renderTodoItems();
    }

    function toggleTodoComplete(itemId) {
      const item = todoItems.find(i => i.id === itemId);
      if (item) {
        item.completed = !item.completed;
        storeSet(STORAGE_KEYS.TODO_ITEMS, todoItems);
        renderTodoGroups();
        renderTodoItems();
      }
    }

    function editTodoItem(itemId) {
      const item = todoItems.find(i => i.id === itemId);
      if (!item) return;
      
      const newContent = prompt('编辑待办内容：', item.content);
      if (newContent === null) return;
      
      item.content = newContent.trim();
      storeSet(STORAGE_KEYS.TODO_ITEMS, todoItems);
      renderTodoGroups();
      renderTodoItems();
    }

    function deleteTodoItem(itemId) {
      if (!confirm('确定删除这条待办？')) return;
      todoItems = todoItems.filter(i => i.id !== itemId);
      storeSet(STORAGE_KEYS.TODO_ITEMS, todoItems);
      renderTodoGroups();
      renderTodoItems();
    }

    function copyTodoContent(itemId) {
      const item = todoItems.find(i => i.id === itemId);
      if (!item) return;
      
      navigator.clipboard.writeText(item.content).then(() => {
        // 简单提示
        const original = event.target.closest('button').innerHTML;
        event.target.closest('button').innerHTML = '<i data-lucide="check" class="w-4 h-4 text-green-400"></i>';
        lucide.createIcons();
        setTimeout(() => {
          event.target.closest('button').innerHTML = '<i data-lucide="copy" class="w-4 h-4"></i>';
          lucide.createIcons();
        }, 1000);
      });
    }

    function exportAllData() {
      const allData = {};
      Object.values(STORAGE_KEYS).forEach(key => {
        allData[key] = storeGet(key, null);
      });
      
      const blob = new Blob([JSON.stringify(allData, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `local-chat-backup-${new Date().toISOString().slice(0,10)}.json`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    function importAllData(event) {
      const file = event.target.files[0];
      if (!file) return;
      
      if (!confirm('导入将覆盖当前所有数据，确定继续吗？')) {
        event.target.value = '';
        return;
      }

      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const data = JSON.parse(e.target.result);
          
          // 写入所有数据
          Object.keys(data).forEach(key => {
            if (data[key] !== null) {
              localStorage.setItem(key, JSON.stringify(data[key]));
            }
          });
          
          alert('导入成功！页面将刷新以加载新数据。');
          location.reload();
        } catch (err) {
          alert('导入失败：文件格式不正确');
        }
      };
      reader.readAsText(file);
      event.target.value = '';
    }

    function clearAllData() {
      if (!confirm('确定要清空所有本地数据吗？此操作不可恢复！')) return;
      if (!confirm('再次确认：所有配置、对话、待办都将被永久删除！')) return;
      
      Object.values(STORAGE_KEYS).forEach(key => localStorage.removeItem(key));
      location.reload();
    }
